cmake_minimum_required(VERSION 3.18)
project(ekvido_core_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(SRC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(GLOB CORE_SOURCES "${SRC_ROOT}/common/ekvido/*.c" "${SRC_ROOT}/common/ekvido/*.cc")
add_executable(ekvido_core_tests test_main.c ekv.c ${CORE_SOURCES})
target_include_directories(ekvido_core_tests PRIVATE "${SRC_ROOT}")
find_package(OpenMP COMPONENTS C CXX)
if(OpenMP_C_FOUND AND OpenMP_CXX_FOUND)
  target_link_libraries(ekvido_core_tests PRIVATE OpenMP::OpenMP_C OpenMP::OpenMP_CXX)
endif()
find_package(Threads REQUIRED)
target_link_libraries(ekvido_core_tests PRIVATE m Threads::Threads)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(ekvido_core_tests PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
endif()

# Speed/memory benchmark (packaging/ekvido/bench-core.sh); same sources and flags as the tests.
add_executable(ekvido_core_bench bench_main.c ${CORE_SOURCES})
target_include_directories(ekvido_core_bench PRIVATE "${SRC_ROOT}")
if(OpenMP_C_FOUND AND OpenMP_CXX_FOUND)
  target_link_libraries(ekvido_core_bench PRIVATE OpenMP::OpenMP_C OpenMP::OpenMP_CXX)
endif()
target_link_libraries(ekvido_core_bench PRIVATE m Threads::Threads)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(ekvido_core_bench PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
endif()
