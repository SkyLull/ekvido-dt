cmake_minimum_required(VERSION 3.18)
project(ekvido_cl_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(SRC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
file(GLOB CORE_SOURCES "${SRC_ROOT}/common/ekvido/*.c" "${SRC_ROOT}/common/ekvido/*.cc")
file(GLOB CL_SOURCES "${SRC_ROOT}/common/ekvido/cl/ekv_cl*.c")
list(FILTER CL_SOURCES EXCLUDE REGEX "ekv_cl_dt\\.c$") # darktable glue: not standalone
set_source_files_properties("${SRC_ROOT}/common/ekvido/cl/ekv_cl_fft.c" PROPERTIES COMPILE_OPTIONS "-w")
find_library(EKV_OPENCL_LIB NAMES OpenCL libOpenCL.so.1 REQUIRED)
find_package(OpenMP COMPONENTS C CXX)
find_package(Threads REQUIRED)
function(ekv_cl_target name)
  add_executable(${name} ${ARGN})
  target_include_directories(${name} PRIVATE "${SRC_ROOT}" "${SRC_ROOT}/external/OpenCL" "${SRC_ROOT}/external/vkfft")
  target_compile_definitions(${name} PRIVATE HAVE_OPENCL=1)
  target_link_libraries(${name} PRIVATE "${EKV_OPENCL_LIB}" m Threads::Threads)
  if(OpenMP_C_FOUND AND OpenMP_CXX_FOUND)
    target_link_libraries(${name} PRIVATE OpenMP::OpenMP_C OpenMP::OpenMP_CXX)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
  endif()
endfunction()
ekv_cl_target(ekvido_cl_tests test_cl_main.c clfn_direct.c ${CL_SOURCES} ${CORE_SOURCES})
ekv_cl_target(vram_ballast vram_ballast.c)
